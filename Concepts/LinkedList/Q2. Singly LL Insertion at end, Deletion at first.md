## Q2. Singly LL Insertion at end, Deletion at first

> Menu driven program with options 1. Insert node at the end 2. Remove a node from the last 3. Display the LL

```java
import java.util.*;

// LinkedList

class Node{
  int data;
  Node next;
  Node(int data){
    this.data=data;
  }
}


// Q2 , menu driven program.. 1. insert a node at the end, 2. remove a node at first
// 3. display the list

class LinkedList{
    Node head;
    Node temp;
    void insert(int data){
        if(head==null){
          Node newNode=new Node(data);
          head=newNode;
          temp=newNode;
        }else{
          Node newNode=new Node(data);
          temp.next=newNode;
          temp=newNode;
        }
    }
    void removeFirst(){
      if(head==null){
        System.out.println("Empty List");
        return;
      }
      head=head.next;
      System.out.println("Removed first element");
    }
    void display(){
      if(head==null){
        System.out.println("Empty List");
        return;
      }
      temp=head;
      while(temp!=null){
        if(temp.next!=null) System.out.print(temp.data+"->");
        else{
          System.out.println(temp.data+"->null");
        }
        temp=temp.next;
      }
    }
}

public class Main {
    public static void main(String[] args) {
      Scanner sc=new Scanner(System.in);
      LinkedList ll=new LinkedList(); // create an obj for LL
      int op=1;
      System.out.println("MENU\n1. insert\n2.remove first\n3.display\n4.exit");
      do{
        System.out.print("Enter choice of op: ");
        op=sc.nextInt();
        switch(op){
          case 1:
            System.out.print("Enter data item: ");
            int data=sc.nextInt();
            ll.insert(data);
            break;
          case 2:
            ll.removeFirst();
            break;
          case 3:
            ll.display();
        }
      }while(op!=4);
    }
}

```
